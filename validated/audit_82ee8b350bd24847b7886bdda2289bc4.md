impl NativeCodeContext {

    pub fn new() -> Self {
        Self {
            enabled: true,
            requested_module_bundle: None,
        }
    }

    pub fn extract_publish_request(&mut self) -> Option<PublishRequest> {
        if !self.enabled {
            return None;
        }

        self.enabled = false;
        self.requested_module_bundle.take()
    }
