## Finding

An unauthenticated caller of the public `auction_submitBid` RPC endpoint can probe arbitrary searcher addresses to determine whether that address has already won an auction slot for a specific target transaction in the upcoming block — before the bid pool validates the searcher's or auctioneer's signatures. This is directly analogous to the Discourse `ComposerController#mentions` bug: a check that reveals privileged/hidden state (group membership → auction-winner identity) is executed and returns a distinguishable response *before* any authorization/signature proof is verified, allowing enumeration by an untrusted caller.

### Root cause

`BidPool.validateBid` performs the "sender already has a different winning bid" check as step 1, well before signature verification (step 6): [1](#0-0) 

Specifically:
- Step 1 (`senderHasDifferentWinner`) returns `auction.ErrBidSenderExists` immediately if `bid.Sender` is already recorded as the winner for `bid.BlockNumber` in `bp.bidWinnerMap`.
- Only in step 6, `validateBidSigs`, are the searcher's EIP-712 signature and the auctioneer's signature actually checked: [2](#0-1) 

Because `bid.Sender` is a plain field of the caller-supplied `BidInput` (not derived from a verified signature) at the point this check runs, an attacker can set it to any address they want to test, submit a bid with garbage `SearcherSig`/`AuctioneerSig`, and distinguish two outcomes:
- `ErrBidSenderExists` → the tested address has already won the block's auction slot.
- Any other error (e.g., invalid signature, invalid block number) → the tested address has not won.

The `SubmitBid` API entry point that reaches this code takes attacker-controlled `BidInput` (including `Sender`) directly from the RPC call, and only needs a syntactically valid, previously-broadcast `targetTxRaw`/`targetTxHash` pair (public mempool data) to reach `AddBid`: [3](#0-2) 

The `senderHasDifferentWinner` helper itself has no authentication/ownership check — it purely looks up the address in an in-memory map: [4](#0-3) 

### Impact

Kaia's KIP-249 auction/MEV mechanism relies on bids and their winners being confidential until the block is sealed (the whole point of the `Auctioneer`-mediated sealed-bid design described in the module's own README). This oracle lets any RPC caller determine, ahead of block inclusion, which competing searcher/address currently holds the winning slot for a specific target transaction, without needing any of the searcher's or auctioneer's private keys. That undermines the confidentiality assumption of the auction, enabling competitors to:
- Detect in real time whether their own bid has been outbid (timing/strategy leak), or
- Enumerate a target competitor's known addresses to learn their live auction standing and adapt bidding strategy accordingly (front-running/counter-bidding the sealed auction), which is a fairness/economic-integrity issue in the MEV auction system.

### Recommendation
Move the signature verification (`validateBidSigs`) ahead of the `senderHasDifferentWinner` check in `BidPool.validateBid`, so that unauthenticated/unsigned probing bids are rejected before any state about existing winners is revealed, mirroring how the Discourse fix reordered checks so unauthorized users cannot reach the membership-revealing code path. [1](#0-0)

// senderHasDifferentWinner reports whether a different bid from the same sender
// already exists in the winner list for this block. Caller must hold bidMu.
func (bp *BidPool) senderHasDifferentWinner(bid *auction.Bid) bool {
	hash, ok := bp.bidWinnerMap[bid.BlockNumber][bid.Sender]
	if !ok {
		return false
	}
	return !bid.Equals(bp.bidMap[hash])
}

func (bp *BidPool) validateBid(bid *auction.Bid) error {
	blockNumber := bid.BlockNumber

	bp.bidMu.RLock()

	// Check if the auction tx is already in the pool.
	if _, ok := bp.bidMap[bid.Hash()]; ok {
		bp.bidMu.RUnlock()
		return auction.ErrBidAlreadyExists
	}

	// 1. The `bid.Sender` must not be in the winner list of the same block number if the new bid isn't equal to the previous bid.
	if bp.senderHasDifferentWinner(bid) {
		bp.bidMu.RUnlock()
		return auction.ErrBidSenderExists
	}
	bp.bidMu.RUnlock()

	curBlock := bp.Chain.CurrentBlock()
	if curBlock == nil {
		return auction.ErrBlockNotFound
	}

	// 2. The `bid.BlockNumber` must be in range of `[currentBlockNumber + 1, currentBlockNumber + allowFutureBlock]`.
	curNum := curBlock.NumberU64()
	if blockNumber <= curNum || blockNumber > curNum+allowFutureBlock {
		return auction.ErrInvalidBlockNumber
	}

	// 3. The `bid.Bid` must be greater than 0.
	if bid.Bid.Sign() <= 0 {
		return auction.ErrZeroBid
	}

	// 4. The data size must be less than the maximum limit.
	if uint64(len(bid.Data)) > BidTxMaxDataSize {
		return auction.ErrExceedMaxDataSize
	}

	// 5. The gas limit must be less than the maximum limit.
	if bid.CallGasLimit > BidTxMaxCallGasLimit {
		return auction.ErrExceedMaxCallGasLimit
	}

	// 6. The `bid.SearcherSig` and `bid.AuctioneerSig` must be valid.
	if err := bp.validateBidSigs(bid); err != nil {
		return err
	}

	return nil
}

func (bp *BidPool) validateBidSigs(bid *auction.Bid) error {
	bp.auctionInfoMu.RLock()
	defer bp.auctionInfoMu.RUnlock()

	if bid.SearcherSig == nil || len(bid.SearcherSig) != crypto.SignatureLength {
		return auction.ErrInvalidSearcherSig
	}
	if bid.AuctioneerSig == nil || len(bid.AuctioneerSig) != crypto.SignatureLength {
		return auction.ErrInvalidAuctioneerSig
	}

	// Verify the EIP712 signature.
	if err := bid.ValidateSearcherSig(bp.ChainConfig.ChainID, bp.auctionEntryPoint, bp.auctionEntryPointVersion); err != nil {
		return err
	}

	// Verify the auctioneer signature.
	if err := bid.ValidateAuctioneerSig(bp.auctioneer); err != nil {
		return err
	}

	return nil
}

func (api *AuctionAPI) SubmitBid(ctx context.Context, bidInput BidInput) RPCOutput {
	numBidRequestCounter.Inc(1)
	if api.a.IsDisabled() {
		return makeRPCOutput(EMPTY_HASH, auction.ErrAuctionDisabled)
	}

	//  1. directly send target transaction
	targetTx, errTxDecode := toTx(bidInput.TargetTxRaw)
	if errTxDecode != nil {
		return makeRPCOutput(EMPTY_HASH, errTxDecode)
	}
	if targetTx.Hash() != bidInput.TargetTxHash {
		return makeRPCOutput(EMPTY_HASH, auction.ErrInvalidTargetTxHash)
	}
	errTargetTxSend := api.a.Backend.SendTx(ctx, targetTx)
	// ignore known transaction related errors against target tx validation
	if errTargetTxSend != nil && !(strings.HasPrefix(errTargetTxSend.Error(), "known transaction:") || errors.Is(errTargetTxSend, gasless_impl.ErrUnableToAddKnownBundleTx)) {
		return makeRPCOutput(EMPTY_HASH, errTargetTxSend)
	}

	// 2. add bid
	bid := ToBid(bidInput)
	bidHash, errValidateBid := api.a.bidPool.AddBid(bid)
	return makeRPCOutput(bidHash, errValidateBid)
}
